Public Function OpenXL() As Boolean
  
  Dim clsCreateRecord As New clsCreateRecord

  Dim i As Integer
  Dim fok As Boolean
  Dim strSQL As String
  Dim strTabName As String
  
  Dim db As Database

  Dim rsGRR_Report As dao.recordSet
  Dim rsgRR_Report_Desc As dao.recordSet
  Dim rsSpacing As dao.recordSet
  Dim rsTab  As dao.recordSet
  Dim rsSpacingTab As dao.recordSet
  Dim rsTableNameHolder As dao.recordSet

  Set db = DBEngine(0)(0)
    
On Error GoTo err_OpenXL:
 
Set rsTab = db.OpenRecordset(prpSqlTab, dbOpenDynaset)
    rsTab.Requery
If rsTab.RecordCount > 0 Then
        rsTab.MoveLast
        rsTab.MoveFirst
        
        
        
      ' objExcel.Workbooks.Open prpFileName, UpdateLinks:=0, ReadOnly:=False
       objExcel.Workbooks.Open prpFileName, ReadOnly = False



    'opend the sheet, maximize the window
    If xlsExcelOpen = True Then
        If prpRender = True Then
        
            objExcel.Application.Visible = True
            objExcel.Application.Windows(1).Visible = True
            objExcel.Application.Windows(1).WindowState = xlMaximized

        Else
        
           objExcel.Application.Visible = False
           'objExcel.Application.Windows(1).Visible = False  'cant use this excel helper window has to remain open
           objExcel.Application.Windows(1).WindowState = xlMinimized
        
            objExcel.Application.ScreenUpdating = False
            objExcel.Application.DisplayAlerts = False
            objExcel.Application.EnableEvents = False
            objExcel.Application.Calculation = xlCalculationManual
            
        End If
    End If
    'create tblNameHolder.  used in recursive function fn_findbreaks()
    If TableOrQueryExists_TSB("", "tblNameHolder") = False Then
                'DoCmd.DeleteObject acTable, prpSortingGrouping
            strSQL = "CREATE TABLE tblNameHolder ([QueryName] text(75), ID int);"
                'class destroys and creates new query
                'Debug.Print strSQL
            With clsCreateRecord
                .prpName = "qryCreateQName"
                .prpSqlString = strSQL
                .prpExecute = True
                fok = .CreateQuery
            End With
    Else
            strSQL = "Delete * From tblNameHolder"
                'class destroys and creates new query
                'Debug.Print strSQL
            With clsCreateRecord
                .prpName = "qryCreateQName"
                .prpSqlString = strSQL
                .prpExecute = True
                fok = .CreateQuery
            End With
    End If
        'tblNameHolder - holds the names of each query/table per tab that contains spacing /grouping info.
         Set rsTableNameHolder = db.OpenRecordset("tblNameHolder", dbOpenDynaset)
         i = 0
    'rsTab - name of the tab of the target spreadsheet
     While rsTab.EOF = False
            
            strSQL = "Delete * From tblNameHolder"
                'class destroys and creates new query
                'Debug.Print strSQL
            With clsCreateRecord
                .prpName = "qryCreateQName"
                .prpSqlString = strSQL
                .prpExecute = True
                fok = .CreateQuery
            End With
            Let prpReportingRules = "tblGoodFieldName_Report"
            
        strTabName = adhReplace(rsTab![Tab], " ", "_") 'simple string replacement - Dev Handbook
        Debug.Print strTabName
        objExcel.Sheets(strTabName).Visible = True
        
            objExcel.Sheets(strTabName).Select 'select tab
                'rsSpacingTab is combo of spacing and tab, figure out the grouping and spacing for the selected sheet/tab
                Set rsSpacingTab = db.OpenRecordset("Select * From " & prpTabSpace & " Where TAB = " & """" & rsTab![Tab] & """", dbOpenSnapshot)
                    rsSpacingTab.Requery
                    If rsSpacingTab.RecordCount > 0 Then
                        rsSpacingTab.Requery
                        rsSpacingTab.MoveFirst
                        'prpReportingRules - tblGoodFieldName - Columns/formating
                        sTab = rsTab![Tab]
                        sReport = rsSpacingTab![Report]
                        bHasTotals = rsSpacingTab![hastotals]
                        Let prpHead_2 = head_2
                        Let prpFoot_2 = foot_2
                        If Not IsNull(rsSpacingTab![over_ride_foot1]) Then
                                Let prpFoot_1 = rsSpacingTab![over_ride_foot1]
                            Else
                                Let prpFoot_1 = foot_1
                        End If
                        If Not IsNull(rsSpacingTab![over_ride_head1]) Then
                                Let prpHead_1 = rsSpacingTab![over_ride_head1]
                            Else
                                Let prpHead_1 = head_1
                        End If
                        Set rsGRR_Report = db.OpenRecordset("SELECT " & prpReportingRules & ".*, * " & _
                            "FROM " & prpReportingRules & " " & _
                            "WHERE (((" & prpReportingRules & ".Tab) = " & """" & sTab & """" & ") " & _
                            "And ((" & prpReportingRules & ".Report) = " & """" & sReport & """" & ")) " & _
                            "ORDER BY " & prpReportingRules & ".Sort;", dbOpenDynaset)
                        Set rsgRR_Report_Desc = db.OpenRecordset("SELECT " & prpReportingRules & ".*, * " & _
                            "FROM " & prpReportingRules & " " & _
                            "WHERE (((" & prpReportingRules & ".Tab) = " & """" & sTab & """" & ") " & _
                            "And ((" & prpReportingRules & ".Report) = " & """" & sReport & """" & ")) " & _
                            "ORDER BY " & prpReportingRules & ".Sort DESC;", dbOpenDynaset)
                        While rsSpacingTab.EOF = False
                           i = i + 1
                           rsTableNameHolder.AddNew
                               rsTableNameHolder![queryname] = rsSpacingTab![queryname]
                               rsTableNameHolder![id] = i
                           rsTableNameHolder.Update
                           rsSpacingTab.MoveNext
                        Wend
                           'call the actual break after all sections (i) have been
                          'calculated
                          rsTableNameHolder.Requery
                           fok = fn_FindBreaks(rsTableNameHolder, "tblNameHolder", 0)
                             If fok = False Then
                                GoTo err_cant_find_break
                               Exit Function
                             End If
                            'we always move first because the primary sort contains all the break data
                             rsTableNameHolder.MoveFirst
                           'call this function for the primary only
                           Call fn_break(rsTableNameHolder![queryname])
                           Call fn_applyFormat(rsGRR_Report, rsgRR_Report_Desc, rsTableNameHolder, 0, bHasTotals)
                           'Call fn_PivotChart(rsGRR_Report, rsgRR_Report_Desc, rsTableNameHolder, 0, strTabName)
                           Call fn_Header(rsGRR_Report, rsgRR_Report_Desc, rsTableNameHolder, 0, strTabName)
                           Call fn_GrandTotalsOnChart(rsGRR_Report, rsgRR_Report_Desc, rsTableNameHolder, 0, strTabName)
                    End If
                        'end find all breaks--------------------------------------------
                rsTab.MoveNext
    Wend
    
                        

                        objExcel.Application.Sheets("Monthly_Chart").Select
                        objExcel.Application.ActiveSheet.Move Before:=objExcel.Application.ActiveWorkbook.Sheets("KPI_Special")

        
        If prpRender = False Then
        
            objExcel.Application.ScreenUpdating = True
            objExcel.Application.DisplayAlerts = True
            objExcel.Application.EnableEvents = True
            objExcel.Application.Calculation = xlCalculationAutomatic
            
            objExcel.Application.Visible = True
            objExcel.Application.Windows(1).Visible = True
            objExcel.Application.Windows(1).WindowState = xlMaximized
            
        Else
            'do nothing
        End If
        
          [Forms]![frmReconAccuracyReviewReports]![txtFolderPathChart].Visible = True
          [Forms]![frmReconAccuracyReviewReports]![txtFolderPathChart].Caption = xlsFileName
          [Forms]![frmReconAccuracyReviewReports]![txtFolderPathChart].HyperlinkAddress = xlsFileName
        
        objExcel.Workbooks.Close
        OpenXL = True
        
         If (rsa.State And adStateOpen) = adStateOpen Then
               rsa.Close
         End If
         
                're-open the workbook?
               ' objExcel.Workbooks.Open prpFileName, ReadOnly = False

     
     Exit Function
Else
    'rstab has no records
    OpenXL = False
    Exit Function
End If
    Exit Function
    
err_cant_find_break:
     MsgBox "Could not create breaks in spreadsheet for sorting and grouping! " & _
     vbCrLf & Err.Number & vbCrLf & _
     Err.Description, vbCritical, "See Developer"
     OpenXL = False
     Exit Function

err_OpenXL:
     MsgBox Err.Number & vbCrLf & Err.Description & vbCrLf & vbCrLf & "Could not format Excel report" & _
     vbCrLf & "Please Contact Developer", vbCritical, "Problem"
     OpenXL = False

End Function
Function fn_applyFormat(ByRef rsGRReport As dao.recordSet, ByRef rsgRR_Report_Desc As dao.recordSet, ByRef rsStrTable As dao.recordSet, i As Long, bHasTotals As Boolean)

Dim intCol As Integer, x As Integer
Dim rssortgroup As dao.recordSet
Dim db As Database
Dim bRk2 As String
'----Grand Total Variables----
Dim rsCount As Long
Dim slvl As String
Dim lSumEnd As Long
Dim lrow As Long
Dim rsSortGroup_GT As dao.recordSet
'----header variables 1-------
Dim vAlignment, vAlignment_2 As Long
Dim hAlignment, hAlignment_2 As Long

Dim lSection As Long

Dim lUnderline, lUnderline_2 As Long
Dim lColorIndex, lColorIndex_2 As Long
Dim bWrapText, bWrapText_2 As Boolean
Dim sFont, sFont_2 As String
Dim sFontStyle, sFontStyle_2 As String

Dim sFormat As String

Dim hEndLetter As String

Dim posAdd_H As Integer
Dim dHeadInsert As String
Dim dHeadInsert2 As String
Dim dHeadInsert3 As String
'----End Header variables-

Set db = DBEngine(0)(0)

If i = 0 Then
iGT = 0
rsGRReport.Requery
rsGRReport.MoveFirst

'clean up-----report is always at A1 need to remove false headers and shift to right--------------------------------------
'insert columns on left to line up report with first column in report.
intCol = ColLet2Num(rsGRReport![Letter]) - ColLet2Num("A")
If intCol > 0 Then
    objExcel.Application.Columns("A:A").Select
    For x = 1 To intCol
        objExcel.Application.Selection.Insert Shift:=xlToRight
    Next x
End If

    bRk2 = "1" & ":" & "1"
    objExcel.Application.Rows(bRk2).Select
    objExcel.Application.Selection.ClearContents
'--------------------------------------------------------------------------------------------------------------------------
 rsStrTable.MoveFirst
 Set rssortgroup = db.OpenRecordset("Select * From " & prpSortingGrouping & " Where " & prpSortingGrouping & ".[Level] = " & """" & rsStrTable![queryname] & """", dbOpenDynaset)
  'apply logic
'-----Begin--sheet-------------------------------------------------------------------------------------------------------
        rsGRReport.MoveFirst
     While rsGRReport.EOF = False
                bRk2 = rsGRReport![Letter] & ":" & rsGRReport![Letter]
                objExcel.Application.Columns(bRk2).ColumnWidth = IIf(IsNull(rsGRReport![ColumnWidth]), 9, rsGRReport![ColumnWidth])
        rsGRReport.MoveNext
     Wend
        rsGRReport.MoveFirst
     While rsGRReport.EOF = False
            If rsGRReport![Column_hide] = True Then
                bRk2 = rsGRReport![Letter] & ":" & rsGRReport![Letter]
                objExcel.Application.Columns(bRk2).Select
                objExcel.Application.Selection.EntireColumn.Hidden = True
            End If
        rsGRReport.MoveNext
     Wend
    '-----------------End-Sheet!--------------------------------------------------------------------------
    'Begin Header-----------------------------------------------------------------------------------------
    '-----------------------------------------------------------------------------------------------------
    '-----------------------------------------------------------------------------------------------------
Else
'skip sheet format
    Set rssortgroup = db.OpenRecordset("Select * From " & prpSortingGrouping & " Where " & prpSortingGrouping & ".[Level] = " & """" & rsStrTable![queryname] & """" & " AND " & prpSortingGrouping & ".[Sort1] = " & rsSort + 1, dbOpenDynaset)
        rssortgroup.Requery
End If

    rssortgroup.MoveFirst
    rssortgroup.MoveLast
    rssortgroup.MoveFirst
    
    rsGRReport.MoveFirst
        
While rssortgroup.EOF = False

While rsGRReport.EOF = False
           'alignment, underline, font color
           Select Case rsGRReport![Alignment_Horiz_H1_A]
                Case "Center"
                   hAlignment = -4108
                Case "Right"
                   hAlignment = -4152
                Case "Left"
                   hAlignment = -4131
                Case Else
                   hAlignment = -4131
           End Select
           Select Case rsGRReport![Alignment_Vert_H1_A]
                Case "Top"
                    vAlignment = -4160
                Case "Center"
                    vAlignment = -4108
                Case "Bottom"
                    vAlignment = -4107
                Case Else
                    vAlignment = -4108
           End Select
           Select Case rsGRReport![Underline_h1_A]
                Case True
                    lUnderline = 2
                Case False
                    lUnderline = -4142
           End Select
           Select Case rsGRReport![Font_Color_H1_A]
                Case "Black"
                    lColorIndex = -4105
                Case "Red"
                    lColorIndex = 3
                Case "Green"
                    lColorIndex = 50
                Case "Blue"
                    lColorIndex = 5
                Case "Gold"
                    lColorIndex = 44
                Case "Light Brown"
                    lColorIndex = 40
                Case "Light Gold"
                    lColorIndex = 36
                Case Else
                    lColorIndex = -4105
           End Select
           Select Case rsGRReport![Underline_h1_AD]
                Case True
                    lUnderline_2 = 2
                Case False
                    lUnderline_2 = -4142
           End Select
           Select Case rsGRReport![Font_Color_H1_AD]
                Case "Black"
                    lColorIndex_2 = -4105
                Case "Red"
                    lColorIndex_2 = 3
                Case "Green"
                    lColorIndex_2 = 50
                Case "Blue"
                    lColorIndex_2 = 5
                Case "Gold"
                    lColorIndex_2 = 44
                Case "Light Brown"
                    lColorIndex = 40
                Case "Light Gold"
                    lColorIndex_2 = 36
                Case Else
                    lColorIndex_2 = -4105
           End Select
'---------------------------------------------------------------------------------------------------------------------
'---------------------------------------------------------------------------------------------------------------------
        'insert data for header  section 1
        dHeadInsert = IIf(IsNull(rsGRReport![fieldname_h1_A]), "", rsGRReport![fieldname_h1_A]) 'evaluate equation
        posAdd_H = IIf(IsNull(rsGRReport![position_h1_a]), 0, rsGRReport![position_h1_a])
        sFont = IIf(IsNull(rsGRReport![font_h1_a]), "", rsGRReport![font_h1_a])
        sFont_2 = IIf(IsNull(rsGRReport![font_h1_ad]), "", rsGRReport![font_h1_ad])
        sFontStyle = IIf(IsNull(rsGRReport![font_style_h1_a]), "", rsGRReport![font_style_h1_a])
        sFontStyle_2 = IIf(IsNull(rsGRReport![font_style_h1_ad]), "", rsGRReport![font_style_h1_ad])
        dFontSize = IIf(IsNull(rsGRReport![font_size_h1_a]), 10, rsGRReport![font_size_h1_a])
        dFontSize_2 = IIf(IsNull(rsGRReport![font_size_h1_ad]), 10, rsGRReport![font_size_h1_ad])
        bWrapText = IIf(IsNull(rsGRReport![wrap_text_h1_a]), False, rsGRReport![wrap_text_h1_a])
        bWrapText_2 = IIf(IsNull(rsGRReport![wrap_text_h1_ad]), False, rsGRReport![wrap_text_h1_ad])
        sFormat = IIf(IsNull(rsGRReport![format_H1_a]), "General", rsGRReport![format_H1_a])
        sFormat_2 = IIf(IsNull(rsGRReport![format_H1_ad]), "General", rsGRReport![format_H1_ad])

                If rsGRReport![HeaderCol_H1_a] = True Then
                    If dHeadInsert <> "" Then
                        'add 1 then add 2
                       objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H), ColLet2Num(rsGRReport![Letter])).Value = _
                       dHeadInsert
                       bRk2 = rsGRReport![Letter] & (rssortgroup![head_B_1] + posAdd_H) & ":" & _
                                rsGRReport![Letter] & (rssortgroup![head_B_1] + posAdd_H)
                            objExcel.Application.Range(bRk2).Select
                            With objExcel.Application.Selection
                                .HorizontalAlignment = hAlignment
                                .VerticalAlignment = vAlignment
                                .WrapText = bWrapText
                            End With
                            With objExcel.Application.Selection.Font
                                .Name = sFont
                                .FontStyle = sFontStyle
                                .Size = dFontSize
                                .Underline = lUnderline
                                .ColorIndex = lColorIndex
                            End With
                            objExcel.Application.Selection.NumberFormat = sFormat
                            
                       objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H) + 1, ColLet2Num(rsGRReport![Letter])) = _
                       objExcel.ActiveSheet.Cells(rssortgroup![begin1], ColLet2Num(rsGRReport![Add_Name_From_Data_H1_A]))
                       bRk2 = rsGRReport![Letter] & (rssortgroup![head_B_1] + posAdd_H) + 1 & ":" & _
                                rsGRReport![Letter] & (rssortgroup![head_B_1] + posAdd_H) + 1
                            objExcel.Application.Range(bRk2).Select
                            With objExcel.Application.Selection
                                .HorizontalAlignment = hAlignment
                                .VerticalAlignment = vAlignment
                                .WrapText = bWrapText_2
                            End With
                            With objExcel.Application.Selection.Font
                                .Name = sFont_2
                                .FontStyle = sFontStyle_2
                                .Size = dFontSize_2
                                .Underline = lUnderline_2
                                .ColorIndex = lColorIndex_2
                            End With
                            objExcel.Application.Selection.NumberFormat = sFormat_2
                        Else
                        'just use from data
                       objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H), ColLet2Num(rsGRReport![Letter])) = _
                       objExcel.ActiveSheet.Cells(rssortgroup![begin1], ColLet2Num(rsGRReport![Add_Name_From_Data_H1_A]))
                       bRk2 = rsGRReport![Letter] & (rssortgroup![head_B_1] + posAdd_H) & ":" & _
                                rsGRReport![Letter] & (rssortgroup![head_B_1] + posAdd_H)
                            objExcel.Application.Range(bRk2).Select
                            With objExcel.Application.Selection
                                .HorizontalAlignment = hAlignment
                                .VerticalAlignment = vAlignment
                                .WrapText = bWrapText_2
                            End With
                            With objExcel.Application.Selection.Font
                                .Name = sFont_2
                                .FontStyle = sFontStyle_2
                                .Size = dFontSize_2
                                .Underline = lUnderline_2
                                .ColorIndex = lColorIndex_2
                            End With
                             objExcel.Application.Selection.NumberFormat = sFormat_2
                    End If
                Else
                 'use dHeadInsert
                       objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H), ColLet2Num(rsGRReport![Letter])).Value = _
                       dHeadInsert
                       bRk2 = rsGRReport![Letter] & (rssortgroup![head_B_1] + posAdd_H) & ":" & _
                                rsGRReport![Letter] & (rssortgroup![head_B_1] + posAdd_H)
                            objExcel.Application.Range(bRk2).Select
                            With objExcel.Application.Selection
                                .HorizontalAlignment = hAlignment
                                .VerticalAlignment = vAlignment
                                .WrapText = bWrapText
                            End With
                            With objExcel.Application.Selection.Font
                                .Name = sFont
                                .FontStyle = sFontStyle
                                .Size = dFontSize
                                .Underline = lUnderline
                                .ColorIndex = lColorIndex
                            End With
                             objExcel.Application.Selection.NumberFormat = sFormat
               End If
                rsGRReport.MoveNext
            Wend
'----------------------------------------------------------------------------------------------------------------------------------------------------------
''---------------------END DATA RETRIVAL FOR SECTIONS FOR HEADER ------------------------------------------------------------------------------------------
'---------------------If they have mergeCells selected then do so - MERGE DATA FIRST -  SECTIONS-----------------------------------------------------------
''---------------------------------------------------------------------------------------------------------------------------------------------------------
      dHeadInsert = ""
      dHeadInsert2 = ""
                rsGRReport.MoveFirst
                While rsGRReport.EOF = False
                      If rsGRReport![Column_hide] = False Then
                        hBeginLetter = rsGRReport![Letter]
                        rsGRReport.MoveLast
                       End If
                    rsGRReport.MoveNext
                Wend
             rsgRR_Report_Desc.MoveFirst
             While rsgRR_Report_Desc.EOF = False
                 If rsgRR_Report_Desc![Column_hide] = False Then
                     hEndLetter = rsgRR_Report_Desc![Letter]
                     rsgRR_Report_Desc.MoveLast
                 End If
                 rsgRR_Report_Desc.MoveNext
             Wend
           rsGRReport.MoveFirst
           If rsGRReport![Merge_Center_H1] = True Then
            While rsGRReport.EOF = False
                'labels
                If rsGRReport![HeaderCol_H1_a] = True Then
                    posAdd_H = IIf(IsNull(rsGRReport![position_h1_a]), 0, rsGRReport![position_h1_a])
                    dHeadInsert = dHeadInsert & "  " & _
                    objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H), ColLet2Num(rsGRReport![Letter]))
                    objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H), ColLet2Num(rsGRReport![Letter])).ClearContents
                    'data
                    dHeadInsert2 = dHeadInsert2 & "  " & _
                    objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H + 1), ColLet2Num(rsGRReport![Letter]))
                    objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H + 1), ColLet2Num(rsGRReport![Letter])).ClearContents
                 End If
              rsGRReport.MoveNext
            Wend
            
            objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H), ColLet2Num(hBeginLetter)) = Trim(dHeadInsert)
            objExcel.ActiveSheet.Cells((rssortgroup![head_B_1] + posAdd_H + 1), ColLet2Num(hBeginLetter)) = Trim(dHeadInsert2)
                
                bRk2 = hBeginLetter & (rssortgroup![head_B_1] + posAdd_H) & ":" & _
                         hEndLetter & (rssortgroup![head_B_1] + posAdd_H)
                     objExcel.Application.Range(bRk2).Select
                       With objExcel.Application.Selection
                         .HorizontalAlignment = -4108
                         .VerticalAlignment = -4108
                         .WrapText = True
                         .MergeCells = True
                      End With

Function ColLet2Num(ColumnLetter As String) As Long
Dim CL As String, CN As Long, n As Integer, S As Long
Dim Se As Integer, c As Integer, T As String, ti As Integer, A As Long

Let CL = ColumnLetter
Let n = Len(CL)
Let S = 0
    For Se = 0 To n - 1
        Let S = S + 26 ^ Se
    Next Se
    Let T = ""
    Let A = 0
    For c = 1 To n
            T = UCase(Mid(CL, c, 1))
            Let ti = 65
            Do Until Chr(ti) = T
                Let ti = ti + 1
                If ti > 90 Then
                    Let CN = "Error"
                    GoTo 1
                End If
            Loop
        Let A = A + ((ti - 65) * (26 ^ (n - c)))
    Next
    Let CN = S + A
1     ColLet2Num = CN
End Function
