
---- Synthetic stored procedure similar in structure and complexity to those I developed at a large insurance company.

CREATE PROCEDURE dbo.usp_ProcessEntityBatch
    @BatchNumber INT,
    @CutoffDate DATETIME
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRANSACTION;

    -------------------------------------------------------------------------
    -- Step 1: Insert selected records from archive into active table
    -------------------------------------------------------------------------
    INSERT INTO ActiveEntityRecords (
        BatchID,
        RegionID,
        GroupID,
        RunDate,
        EntityNumber,
        RecordType,
        SubGroupID,
        BaseGroupID,
        SubscriberID,
        PersonIdentifier,
        OldPersonIdentifier,
        EmployeeNumber,
        HireDate,
        EffectiveDate,
        CancelDate,
        FamilySize,
        MedicalPremium,
        DentalPremium,
        AccessFlag,
        CancelReasonCode,
        AddReasonCode,
        EmploymentStatusCode,
        OldMemberID,
        PovertyLevelCode,
        HouseholdNumber,
        IncomeAmount,
        ProcessFlag,
        MemberRecordFlag,
        ApplicationDate
    )
    SELECT
        a.BatchID,
        a.RegionID,
        a.GroupID,
        a.RunDate,
        a.EntityNumber,
        a.RecordType,
        a.SubGroupID,
        a.BaseGroupID,
        a.SubscriberID,
        a.PersonIdentifier,
        a.OldPersonIdentifier,
        a.EmployeeNumber,
        a.HireDate,
        a.EffectiveDate,
        a.CancelDate,
        a.FamilySize,
        a.MedicalPremium,
        a.DentalPremium,
        a.AccessFlag,
        a.CancelReasonCode,
        a.AddReasonCode,
        a.EmploymentStatusCode,
        a.OldMemberID,
        a.PovertyLevelCode,
        a.HouseholdNumber,
        a.IncomeAmount,
        a.ProcessFlag,
        a.MemberRecordFlag,
        a.ApplicationDate
    FROM ArchiveEntityRecords a
    WHERE a.BatchID IN (
        SELECT BatchID
        FROM BatchQueueLog
        WHERE QueueState = @BatchNumber
          AND ChangeDate > @CutoffDate
    );

    -------------------------------------------------------------------------
    -- Step 2: Update run date
    -------------------------------------------------------------------------
    UPDATE ActiveEntityRecords
    SET RunDate = CONVERT(VARCHAR(8), GETDATE(), 112)
    WHERE RunDate = CONVERT(VARCHAR(8), DATEADD(DAY, -1, GETDATE()), 112);

    -------------------------------------------------------------------------
    -- Step 3: Delete processed archive records
    -------------------------------------------------------------------------
    DELETE FROM ArchiveEntityRecords
    WHERE BatchID IN (
        SELECT BatchID
        FROM BatchQueueLog
        WHERE QueueState = @BatchNumber
          AND ChangeDate > @CutoffDate
    );

    -------------------------------------------------------------------------
    -- Step 4: Insert member-level records from archive
    -------------------------------------------------------------------------
    INSERT INTO ActiveMemberRecords (
        BatchID,
        RegionID,
        GroupID,
        RunDate,
        EntityNumber,
        RecordType,
        MemberNumber,
        SubGroupID,
        BaseGroupID,
        SubscriberID,
        PersonHealthID,
        SubRecordFlag,
        RelationCode,
        DisplayName,
        GenderCode,
        PersonIdentifier,
        OldPersonIdentifier,
        DateOfBirth,
        AddressLine1,
        AddressLine2,
        AddressLine3,
        ZipCode,
        ZipPlus4,
        City,
        StateID,
        PhoneNumber,
        PhoneExtension,
        CountyID,
        CountryID,
        WorkPhone,
        WorkExtension,
        MemberEffectiveDate,
        MemberCancelDate,
        DependentStatusCode,
        MedicareID,
        MedicarePlanID,
        MedicareEntitlementCode,
        MedicareEffectiveDate,
        PartAEffectiveDate,
        PartBEffectiveDate,
        DisabilityFlag,
        PartAAssignDate,
        PartBAssignDate,
        FacilityID,
        FacilityEffectiveDate,
        FacilityEndDate,
        ProviderID,
        ProviderEffectiveDate,
        ProviderEndDate,
        ProviderChangeReasonCode,
        CancelReasonCode,
        AddReasonCode,
        OldMemberSystemID,
        UnbundleCode,
        UnbundlePercent,
        UnbundleEffectiveDate,
        ChoiceFlag,
        MailAddressLine1,
        MailAddressLine2,
        MailCity,
        MailStateCode,
        MailZipCode,
        MailZipPlus4,
        MedicareSignDate,
        MedicareDeletionDate,
        ESRDFlag,
        COBFlag,
        SecondaryRxID,
        SecondaryRxGroup,
        ProcessFlag,
        AdditionalSubscriberID,
        AdditionalMemberID,
        EmployerID,
        EmployerAltID,
        SubsidyFlag,
        SubsidyEffectiveDate,
        TobaccoFlag,
        TobaccoEffectiveDate,
        EthnicityCode,
        MaritalStatusCode,
        PrimaryLanguageCode,
        EmailAddress,
        ReservedField1,
        ReservedField2,
        ReservedField3
    )
    SELECT
        m.BatchID,
        m.RegionID,
        m.GroupID,
        m.RunDate,
        m.EntityNumber,
        m.RecordType,
        m.MemberNumber,
        m.SubGroupID,
        m.BaseGroupID,
        m.SubscriberID,
        m.PersonHealthID,
        m.SubRecordFlag,
        m.RelationCode,
        m.DisplayName,
        m.GenderCode,
        m.PersonIdentifier,
        m.OldPersonIdentifier,
        m.DateOfBirth,
        m.AddressLine1,
        m.AddressLine2,
        m.AddressLine3,
        m.ZipCode,
        m.ZipPlus4,
        m.City,
        m.StateID,
        m.PhoneNumber,
        m.PhoneExtension,
        m.CountyID,
        m.CountryID,
        m.WorkPhone,
        m.WorkExtension,
        m.MemberEffectiveDate,
        m.MemberCancelDate,
        m.DependentStatusCode,
        m.MedicareID,
        m.MedicarePlanID,
        m.MedicareEntitlementCode,
        m.MedicareEffectiveDate,
        m.PartAEffectiveDate,
        m.PartBEffectiveDate,
        m.DisabilityFlag,
        m.PartAAssignDate,
        m.PartBAssignDate,
        m.FacilityID,
        m.FacilityEffectiveDate,
        m.FacilityEndDate,
        m.ProviderID,
        m.ProviderEffectiveDate,
        m.ProviderEndDate,
        m.ProviderChangeReasonCode,
        m.CancelReasonCode,
        m.AddReasonCode,
        m.OldMemberSystemID,
        m.UnbundleCode,
        m.UnbundlePercent,
        m.UnbundleEffectiveDate,
        m.ChoiceFlag,
        m.MailAddressLine1,
        m.MailAddressLine2,
        m.MailCity,
        m.MailStateCode,
        m.MailZipCode,
        m.MailZipPlus4,
        m.MedicareSignDate,
        m.MedicareDeletionDate,
        m.ESRDFlag,
        m.COBFlag,
        m.SecondaryRxID,
        m.SecondaryRxGroup,
        m.ProcessFlag,
        m.AdditionalSubscriberID,
        m.AdditionalMemberID,
        m.EmployerID,
        m.EmployerAltID,
        m.SubsidyFlag,
        m.SubsidyEffectiveDate,
        m.TobaccoFlag,
        m.TobaccoEffectiveDate,
        m.EthnicityCode,
        m.MaritalStatusCode,
        m.PrimaryLanguageCode,
        m.EmailAddress,
        m.ReservedField1,
        m.ReservedField2,
        m.ReservedField3
    FROM ArchiveMemberRecords m
    WHERE m.BatchID IN (
        SELECT BatchID
        FROM BatchQueueLog
        WHERE QueueState = @BatchNumber
          AND ChangeDate > @CutoffDate
    );

    -------------------------------------------------------------------------
    -- Step 5: Update run date for member records
    -------------------------------------------------------------------------
    UPDATE ActiveMemberRecords
    SET RunDate = CONVERT(VARCHAR(8), GETDATE(), 112)
    WHERE RunDate = CONVERT(VARCHAR(8), DATEADD(DAY, -1, GETDATE()), 112);

    -------------------------------------------------------------------------
    -- Step 6: Delete processed member archive records
    -------------------------------------------------------------------------
    DELETE FROM ArchiveMemberRecords
    WHERE BatchID IN (
        SELECT BatchID
        FROM BatchQueueLog
        WHERE QueueState = @BatchNumber
          AND ChangeDate > @CutoffDate
    );

    COMMIT TRANSACTION;

    -------------------------------------------------------------------------
    -- Step 7: Reset queue state
    -------------------------------------------------------------------------
    BEGIN TRANSACTION;
        UPDATE BatchQueue
        SET QueueState = 100
        WHERE BatchID IN (
            SELECT BatchID
            FROM BatchQueueLog
            WHERE QueueState = @BatchNumber
              AND ChangeDate > @CutoffDate
        );
    COMMIT TRANSACTION;

    -------------------------------------------------------------------------
    -- Step 8: Return affected queue rows
    -------------------------------------------------------------------------
    SELECT *
    FROM BatchQueue
    WHERE BatchID IN (
        SELECT BatchID
        FROM BatchQueueLog
        WHERE QueueState = @BatchNumber
          AND ChangeDate > @CutoffDate
    );

END;
