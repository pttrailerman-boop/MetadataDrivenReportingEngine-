sql
-- Synthetic stored procedure similar in structure and complexity to those I developed at a large insurance company.
-- All table names, column names, and business logic have been fully sanitized.

CREATE PROCEDURE dbo.usp_ScoreEntityMatch
(
    @RegionCode      CHAR(2),
    @GroupCode       VARCHAR(10),
    @InputSSN        VARCHAR(9),
    @InputEmployeeID VARCHAR(12),
    @InputFullName   VARCHAR(50),
    @InputDOB        VARCHAR(8),
    @InputZIP        VARCHAR(5),
    @InputGender     CHAR(1),
    @InputPhone      VARCHAR(11),
    @Threshold       INT
)
AS
BEGIN
    SET NOCOUNT ON;

    -------------------------------------------------------------------------
    -- Weight configuration (synthetic values)
    -------------------------------------------------------------------------
    DECLARE @WeightSSN          INT = 10;
    DECLARE @WeightFullName     INT = 15;
    DECLARE @WeightLastName     INT = 5;
    DECLARE @WeightFirstName    INT = 5;
    DECLARE @WeightDOBExact     INT = 20;
    DECLARE @WeightDOBRange     INT = 15;
    DECLARE @WeightGender       INT = 5;
    DECLARE @WeightZIP          INT = 10;
    DECLARE @WeightPhone        INT = 5;

    -------------------------------------------------------------------------
    -- Parse incoming name into first/last
    -------------------------------------------------------------------------
    DECLARE @FirstName VARCHAR(50);
    DECLARE @LastName  VARCHAR(50);

    SET @InputFullName = LTRIM(RTRIM(@InputFullName));

    -- LastName,FirstName format
    IF CHARINDEX(',', @InputFullName) > 0
    BEGIN
        SET @LastName  = LEFT(@InputFullName, CHARINDEX(',', @InputFullName) - 1);
        SET @FirstName = LTRIM(RTRIM(RIGHT(@InputFullName, LEN(@InputFullName) - CHARINDEX(',', @InputFullName))));
    END
    ELSE
    BEGIN
        SET @LastName  = @InputFullName;
        SET @FirstName = @InputFullName;
    END

    -------------------------------------------------------------------------
    -- DOB range logic (± 1 year in YYYYMMDD integer format)
    -------------------------------------------------------------------------
    DECLARE @DOBExact BIGINT = CAST(@InputDOB AS BIGINT);
    DECLARE @DOBUpper BIGINT = @DOBExact + 10000;
    DECLARE @DOBLower BIGINT = @DOBExact - 10000;

    -------------------------------------------------------------------------
    -- Main fuzzy‑matching query
    -------------------------------------------------------------------------
    SELECT
        EntityID,
        PersonID,
        PersonSSN,
        DisplayName,
        DateOfBirth,
        ZIPCode,
        GenderCode,
        PhoneNumber,

        ---------------------------------------------------------------------
        -- Weighted score calculation
        ---------------------------------------------------------------------
        (
            -- SSN or EmployeeID match
            CASE 
                WHEN @InputSSN = '000000000'
                    THEN CASE WHEN EmployeeID = LTRIM(RTRIM(@InputEmployeeID)) THEN @WeightSSN ELSE 0 END
                ELSE CASE WHEN PersonSSN = CAST(@InputSSN AS INT) THEN @WeightSSN ELSE 0 END
            END

            +

            -- Full name or partial name match
            CASE 
                WHEN LTRIM(RTRIM(DisplayName)) = @InputFullName THEN @WeightFullName
                ELSE
                    CASE 
                        WHEN LEFT(DisplayName, CHARINDEX(',', DisplayName) - 1) = @LastName THEN @WeightLastName ELSE 0 END
                    +
                        CASE 
                            WHEN SUBSTRING(DisplayName, CHARINDEX(',', DisplayName) + 1, LEN(DisplayName)) = @FirstName THEN @WeightFirstName ELSE 0 END
                    END
            END

            +

            -- DOB exact or range match
            CASE 
                WHEN CAST(DateOfBirth AS BIGINT) = @DOBExact THEN @WeightDOBExact
                WHEN CAST(DateOfBirth AS BIGINT) BETWEEN @DOBLower AND @DOBUpper THEN @WeightDOBRange
                ELSE 0
            END

            +

            -- Gender match
            CASE WHEN GenderCode = @InputGender THEN @WeightGender ELSE 0 END

            +

            -- ZIP match
            CASE WHEN ZIPCode = CAST(@InputZIP AS INT) THEN @WeightZIP ELSE 0 END

            +

            -- Phone match
            CASE 
                WHEN @InputPhone = '00000000000'
                    THEN CASE WHEN PhoneNumber <> 0 THEN @WeightPhone ELSE 0 END
                ELSE CASE WHEN PhoneNumber = CAST(RIGHT(@InputPhone, 10) AS INT) THEN @WeightPhone ELSE 0 END
            END
        ) AS MatchScore,

        '1' AS PassFlag

    FROM EntityWarehouse WITH (NOLOCK)
    WHERE RegionCode = @RegionCode
      AND GroupCode  = CAST(@GroupCode AS INT)
      AND StatusCode = 'ACTIVE'
      AND DuplicateFlag = 0

    -------------------------------------------------------------------------
    -- Only return records meeting the threshold
    -------------------------------------------------------------------------
    HAVING
        (
            -- same scoring block as above (synthetic)
            CASE 
                WHEN @InputSSN = '000000000'
                    THEN CASE WHEN EmployeeID = LTRIM(RTRIM(@InputEmployeeID)) THEN @WeightSSN ELSE 0 END
                ELSE CASE WHEN PersonSSN = CAST(@InputSSN AS INT) THEN @WeightSSN ELSE 0 END
            END
            +
            CASE 
                WHEN LTRIM(RTRIM(DisplayName)) = @InputFullName THEN @WeightFullName
                ELSE
                    CASE WHEN LEFT(DisplayName, CHARINDEX(',', DisplayName) - 1) = @LastName THEN @WeightLastName ELSE 0 END
                    +
                    CASE WHEN SUBSTRING(DisplayName, CHARINDEX(',', DisplayName) + 1, LEN(DisplayName)) = @FirstName THEN @WeightFirstName ELSE 0 END
                END
            END
            +
            CASE 
                WHEN CAST(DateOfBirth AS BIGINT) = @DOBExact THEN @WeightDOBExact
                WHEN CAST(DateOfBirth AS BIGINT) BETWEEN @DOBLower AND @DOBUpper THEN @WeightDOBRange
                ELSE 0
            END
            +
            CASE WHEN GenderCode = @InputGender THEN @WeightGender ELSE 0 END
            +
            CASE WHEN ZIPCode = CAST(@InputZIP AS INT) THEN @WeightZIP ELSE 0 END
            +
            CASE 
                WHEN @InputPhone = '00000000000'
                    THEN CASE WHEN PhoneNumber <> 0 THEN @WeightPhone ELSE 0 END
                ELSE CASE WHEN PhoneNumber = CAST(RIGHT(@InputPhone, 10) AS INT) THEN @WeightPhone ELSE 0 END
            END
        ) >= @Threshold;

END;
