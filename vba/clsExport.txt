Public Function OpenXL() As Boolean
  
  Dim clsCreateRecord As New clsCreateRecord

  Dim i As Integer
  Dim fok As Boolean
  Dim strSQL As String
  Dim strTabName As String
  
  Dim db As Database

  Dim rsGRR_Report As DAO.Recordset
  Dim rsgRR_Report_Desc As DAO.Recordset
  Dim rsSpacing As DAO.Recordset
  Dim rsTab  As DAO.Recordset
  Dim rsSpacingTab As DAO.Recordset
  Dim rsTableNameHolder As DAO.Recordset


  Set db = DBEngine(0)(0)
    
On Error GoTo err_OpenXL:
 
Set rsTab = db.OpenRecordset(prpSqlTab, dbOpenDynaset)
    rsTab.Requery
If rsTab.RecordCount > 0 Then
        rsTab.MoveLast
        rsTab.MoveFirst
        
        
        
      ' objExcel.Workbooks.Open prpFileName, UpdateLinks:=0, ReadOnly:=False
       objExcel.Workbooks.Open prpFileName, ReadOnly = False



    'opend the sheet, maximize the window
    If xlsExcelOpen = True Then
        If prpRender = True Then
        
            objExcel.Application.Visible = True
            objExcel.Application.Windows(1).Visible = True
            objExcel.Application.Windows(1).WindowState = xlMaximized

        Else
        
           objExcel.Application.Visible = False
           'objExcel.Application.Windows(1).Visible = False  'cant use this excel helper window has to remain open
           objExcel.Application.Windows(1).WindowState = xlMinimized
        
            objExcel.Application.ScreenUpdating = False
            objExcel.Application.DisplayAlerts = False
            objExcel.Application.EnableEvents = False
            objExcel.Application.Calculation = xlCalculationManual
            
        End If
    End If
    'create tblNameHolder.  used in recursive function fn_findbreaks()
    If TableOrQueryExists_TSB("", "tblNameHolder") = False Then
                'DoCmd.DeleteObject acTable, prpSortingGrouping
            strSQL = "CREATE TABLE tblNameHolder ([QueryName] text(75), ID int);"
                'class destroys and creates new query
                'Debug.Print strSQL
            With clsCreateRecord
                .prpName = "qryCreateQName"
                .prpSqlString = strSQL
                .prpExecute = True
                fok = .CreateQuery
            End With
    Else
            strSQL = "Delete * From tblNameHolder"
                'class destroys and creates new query
                'Debug.Print strSQL
            With clsCreateRecord
                .prpName = "qryCreateQName"
                .prpSqlString = strSQL
                .prpExecute = True
                fok = .CreateQuery
            End With
    End If
        'tblNameHolder - holds the names of each query/table per tab that contains spacing /grouping info.
         Set rsTableNameHolder = db.OpenRecordset("tblNameHolder", dbOpenDynaset)
         i = 0
    'rsTab - name of the tab of the target spreadsheet
     While rsTab.EOF = False
            
            strSQL = "Delete * From tblNameHolder"
                'class destroys and creates new query
                'Debug.Print strSQL
            With clsCreateRecord
                .prpName = "qryCreateQName"
                .prpSqlString = strSQL
                .prpExecute = True
                fok = .CreateQuery
            End With
            Let prpReportingRules = "tblGoodFieldName_Report"
            
        strTabName = adhReplace(rsTab![Tab], " ", "_") 'simple string replacement - Dev Handbook
        Debug.Print strTabName
        objExcel.Sheets(strTabName).Visible = True
        
            objExcel.Sheets(strTabName).Select 'select tab
                'rsSpacingTab is combo of spacing and tab, figure out the grouping and spacing for the selected sheet/tab
                Set rsSpacingTab = db.OpenRecordset("Select * From " & prpTabSpace & " Where TAB = " & """" & rsTab![Tab] & """", dbOpenSnapshot)
                    rsSpacingTab.Requery
                    If rsSpacingTab.RecordCount > 0 Then
                        rsSpacingTab.Requery
                        rsSpacingTab.MoveFirst
                        'prpReportingRules - tblGoodFieldName - Columns/formating
                        sTab = rsTab![Tab]
                        sReport = rsSpacingTab![Report]
                        bHasTotals = rsSpacingTab![hastotals]
                        Let prpHead_2 = head_2
                        Let prpFoot_2 = foot_2
                        If Not IsNull(rsSpacingTab![over_ride_foot1]) Then
                                Let prpFoot_1 = rsSpacingTab![over_ride_foot1]
                            Else
                                Let prpFoot_1 = foot_1
                        End If
                        If Not IsNull(rsSpacingTab![over_ride_head1]) Then
                                Let prpHead_1 = rsSpacingTab![over_ride_head1]
                            Else
                                Let prpHead_1 = head_1
                        End If
                        Set rsGRR_Report = db.OpenRecordset("SELECT " & prpReportingRules & ".*, * " & _
                            "FROM " & prpReportingRules & " " & _
                            "WHERE (((" & prpReportingRules & ".Tab) = " & """" & sTab & """" & ") " & _
                            "And ((" & prpReportingRules & ".Report) = " & """" & sReport & """" & ")) " & _
                            "ORDER BY " & prpReportingRules & ".Sort;", dbOpenDynaset)
                        Set rsgRR_Report_Desc = db.OpenRecordset("SELECT " & prpReportingRules & ".*, * " & _
                            "FROM " & prpReportingRules & " " & _
                            "WHERE (((" & prpReportingRules & ".Tab) = " & """" & sTab & """" & ") " & _
                            "And ((" & prpReportingRules & ".Report) = " & """" & sReport & """" & ")) " & _
                            "ORDER BY " & prpReportingRules & ".Sort DESC;", dbOpenDynaset)
                        While rsSpacingTab.EOF = False
                           i = i + 1
                           rsTableNameHolder.AddNew
                               rsTableNameHolder![queryname] = rsSpacingTab![queryname]
                               rsTableNameHolder![id] = i
                           rsTableNameHolder.Update
                           rsSpacingTab.MoveNext
                        Wend
                           'call the actual break after all sections (i) have been
                          'calculated
                          rsTableNameHolder.Requery
                           fok = fn_FindBreaks(rsTableNameHolder, "tblNameHolder", 0)
                             If fok = False Then
                                GoTo err_cant_find_break
                               Exit Function
                             End If
                            'we always move first because the primary sort contains all the break data
                             rsTableNameHolder.MoveFirst
                           'call this function for the primary only
                           Call fn_break(rsTableNameHolder![queryname])
                           Call fn_applyFormat(rsGRR_Report, rsgRR_Report_Desc, rsTableNameHolder, 0, bHasTotals)
                           'Call fn_PivotChart(rsGRR_Report, rsgRR_Report_Desc, rsTableNameHolder, 0, strTabName) example code only
                           Call fn_ReportHeader(rsGRR_Report, rsgRR_Report_Desc, rsTableNameHolder, strTabName)
                           Call CreateMedicareTop10Chart(rsTableNameHolder, strTabName)

                    End If
                        'end find all breaks--------------------------------------------
                rsTab.MoveNext
    Wend

        
        If prpRender = True Then
        
            objExcel.Application.ScreenUpdating = True
            objExcel.Application.DisplayAlerts = True
            objExcel.Application.EnableEvents = True
            objExcel.Application.Calculation = xlCalculationAutomatic
            
            objExcel.Application.Visible = True
            objExcel.Application.Windows(1).Visible = True
            objExcel.Application.Windows(1).WindowState = xlMaximized
            
        Else
            'do nothing
        End If
        

        
        '-------------------------------------------------------------
        'Complete all pending calculations before saving.
        '
        'This prevents Excel from saving formula cells as stale when
        'the report was built with calculation temporarily set to
        'manual for performance.
        '-------------------------------------------------------------
        objExcel.Application.Calculation = xlCalculationAutomatic
        
        objExcel.ActiveWorkbook.ForceFullCalculation = True
        objExcel.Application.CalculateFullRebuild
        
        DoEvents
        
        objExcel.Application.DisplayAlerts = False
        objExcel.ActiveWorkbook.Save
        objExcel.ActiveWorkbook.Close SaveChanges:=False
        objExcel.Application.Quit
        
        OpenXL = True
        
         If (rsa.State And adStateOpen) = adStateOpen Then
               rsa.Close
         End If
         
                're-open the workbook?
               ' objExcel.Workbooks.Open prpFileName, ReadOnly = False

     Exit Function
Else
    'rstab has no records
    OpenXL = False
    Exit Function
End If
    Exit Function
    
err_cant_find_break:
     MsgBox "Could not create breaks in spreadsheet for sorting and grouping! " & _
     vbCrLf & Err.Number & vbCrLf & _
     Err.Description, vbCritical, "See Developer"
     OpenXL = False
     Exit Function

err_OpenXL:
     MsgBox Err.Number & vbCrLf & Err.Description & vbCrLf & vbCrLf & "Could not format Excel report" & _
     vbCrLf & "Please Contact Developer", vbCritical, "Problem"
     OpenXL = False

End Function
Function fn_applyFormat(ByRef rsGRReport As DAO.Recordset, ByRef rsgRR_Report_Desc As DAO.Recordset, ByRef rsStrTable As DAO.Recordset, i As Long, bHasTotals As Boolean)

Dim intCol As Integer, x As Integer
Dim rssortgroup As DAO.Recordset
Dim db As Database
Dim bRk2 As String
'----Grand Total Variables----
Dim rsCount As Long
Dim slvl As String
Dim lSumEnd As Long
Dim lrow As Long
Dim rsSortGroup_GT As DAO.Recordset

'----band header variables 1-------
Dim vAlignment, vAlignment_2 As Long
Dim hAlignment, hAlignment_2 As Long

Dim lSection As Long

Dim lUnderline, lUnderline_2 As Long
Dim lColorIndex, lColorIndex_2 As Long
Dim bWrapText, bWrapText_2 As Boolean
Dim sFont, sFont_2 As String
Dim sFontStyle, sFontStyle_2 As String
Dim dFontSize, dFontSize_2 As Double

Dim lLineWeight As Long
Dim sFormat As String
Dim sFormat_2 As String

Dim strFormula As String

Dim hBeginLetter As String
Dim hBeginLetter_2 As String
Dim hEndLetter As String

Dim posAdd_H As Integer
Dim dHeadInsert As String
Dim dHeadInsert2 As String
Dim dHeadInsert3 As String
'----End Header variables-

Set db = DBEngine(0)(0)

If i = 0 Then
iGT = 0 ' called recursively we would end up re-running grand totals again

rsGRReport.Requery
rsGRReport.MoveFirst

'clean up-----report is always at A1 need to remove false headers and shift to right--------You can set tblGoodFieldName_Report, G_totals, Header to start on any letter--------------
'insert columns on left to line up report with first column in report.   You don't have to start with A Column.
intCol = ColLet2Num(rsGRReport![Letter]) - ColLet2Num("A")
If intCol > 0 Then
    objExcel.Application.Columns("A:A").Select
    For x = 1 To intCol
        objExcel.Application.Selection.Insert Shift:=xlToRight
    Next x
End If

    bRk2 = "1" & ":" & "1"
    objExcel.Application.Rows(bRk2).Select
    objExcel.Application.Selection.ClearContents
'--------------------------------------------------------------------------------------------------------------------------
 rsStrTable.MoveFirst
 Set rssortgroup = db.OpenRecordset("Select * From " & prpSortingGrouping & " Where " & prpSortingGrouping & ".[Level] = " & """" & rsStrTable![queryname] & """", dbOpenDynaset)
  'apply logic
'-----Begin--sheet-------------------------------------------------------------------------------------------------------
        rsGRReport.MoveFirst
     While rsGRReport.EOF = False
                bRk2 = rsGRReport![Letter] & ":" & rsGRReport![Letter]
                objExcel.Application.Columns(bRk2).ColumnWidth = IIf(IsNull(rsGRReport![ColumnWidth]), 9, rsGRReport![ColumnWidth])
        rsGRReport.MoveNext
     Wend
        rsGRReport.MoveFirst
     While rsGRReport.EOF = False
            If rsGRReport![Column_hide] = True Then
                bRk2 = rsGRReport![Letter] & ":" & rsGRReport![Letter]
                objExcel.Application.Columns(bRk2).Select
                objExcel.Application.Selection.EntireColumn.Hidden = True
            End If
        rsGRReport.MoveNext
     Wend
    '-----------------End-Sheet!--------------------------------------------------------------------------
    'Begin Header-----------------------------------------------------------------------------------------
    '-----------------------------------------------------------------------------------------------------
    '-----------------------------------------------------------------------------------------------------
Else
'skip sheet format
    Set rssortgroup = db.OpenRecordset("Select * From " & prpSortingGrouping & " Where " & prpSortingGrouping & ".[Level] = " & """" & rsStrTable![queryname] & """" & " AND " & prpSortingGrouping & ".[Sort1] = " & rsSort + 1, dbOpenDynaset)
        rssortgroup.Requery
End If

    rssortgroup.MoveFirst
    rssortgroup.MoveLast
    rssortgroup.MoveFirst
    
    rsGRReport.MoveFirst
        
While rssortgroup.EOF = False

While rsGRReport.EOF = False
           'alignment, underline, font color
           Select Case rsGRReport![Alignment_Horiz_H1_A]
                Case "Center"
                   hAlignment = -4108
                Case "Right"
                   hAlignment = -4152
                Case "Left"
                   hAlignment = -4131
                Case Else
                   hAlignment = -4131
           End Select
           Select Case rsGRReport![Alignment_Vert_H1_A]
                Case "Top"
                    vAlignment = -4160
                Case "Center"
                    vAlignment = -4108
                Case "Bottom"
                    vAlignment = -4107
                Case Else
                    vAlignment = -4108
           End Select
           Select Case rsGRReport![Underline_H1_A]
                Case True
                    lUnderline = 2
                Case False
                    lUnderline = -4142
           End Select
           Select Case rsGRReport![Font_Color_H1_A]
                Case "Black"
                    lColorIndex = -4105
                Case "Red"
                    lColorIndex = 3
                Case "Green"
                    lColorIndex = 50
                Case "Blue"
                    lColorIndex = 5
                Case "Gold"
                    lColorIndex = 44
                Case "Light Brown"
                    lColorIndex = 40
                Case "Light Gold"
                    lColorIndex = 36
                Case Else
                    lColorIndex = -4105
           End Select
           Select Case rsGRReport![Underline_h1_AD]
                Case True
                    lUnderline_2 = 2
                Case False
                    lUnderline_2 = -4142
           End Select
           Select Case rsGRReport![Font_Color_H1_AD]
                Case "Black"
                    lColorIndex_2 = -4105
                Case "Red"
                    lColorIndex_2 = 3
                Case "Green"
                    lColorIndex_2 = 50
                Case "Blue"
                    lColorIndex_2 = 5
                Case "Gold"
                    lColorIndex_2 = 44
                Case "Light Brown"
                    lColorIndex = 40
                Case "Light Gold"
                    lColorIndex_2 = 36
                Case Else
                    lColorIndex_2 = -4105
           End Select
'---------------------------------------------------------------------------------------------------------------------
'---------------------------------------------------------------------------------------------------------------------
        'insert data for header  section 1
        dHeadInsert = IIf(IsNull(rsGRReport![FieldName_H1_A]), "", rsGRReport![FieldName_H1_A]) 'evaluate equation
        posAdd_H = IIf(IsNull(rsGRReport![position_h1_a]), 0, rsGRReport![position_h1_a])
        sFont = IIf(IsNull(rsGRReport![Font_H1_A]), "", rsGRReport![Font_H1_A])
        sFont_2 = IIf(IsNull(rsGRReport![font_h1_ad]), "", rsGRReport![font_h1_ad])
        sFontStyle = IIf(IsNull(rsGRReport![Font_Style_H1_A]), "", rsGRReport![Font_Style_H1_A])
        sFontStyle_2 = IIf(IsNull(rsGRReport![font_style_h1_ad]), "", rsGRReport![font_style_h1_ad])
        dFontSize = IIf(IsNull(rsGRReport![Font_Size_H1_A]), 10, rsGRReport![Font_Size_H1_A])
        dFontSize_2 = IIf(IsNull(rsGRReport![font_size_h1_ad]), 10, rsGRReport![font_size_h1_ad])
        bWrapText = IIf(IsNull(rsGRReport![Wrap_Text_H1_A]), False, rsGRReport![Wrap_Text_H1_A])
        bWrapText_2 = IIf(IsNull(rsGRReport![wrap_text_h1_ad]), False, rsGRReport![wrap_text_h1_ad])
        sFormat = IIf(IsNull(rsGRReport![Format_H1_A]), "General", rsGRReport![Format_H1_A])
        sFormat_2 = IIf(IsNull(rsGRReport![format_H1_ad]), "General", rsGRReport![format_H1_ad])

                If rsGRReport![HeaderCol_H1_a] = True Then
                    If dHeadInsert <> "" Then
                        'add 1 then add 2
                       objExcel.ActiveSheet.Cells((CLng(rssortgroup![Head_b_1] + posAdd_H)), ColLet2Num(rsGRReport![Letter])).Value = _
                       dHeadInsert
                       bRk2 = rsGRReport![Letter] & (CLng(rssortgroup![Head_b_1]) + posAdd_H) & ":" & _
                                rsGRReport![Letter] & (CLng(rssortgroup![Head_b_1]) + posAdd_H)
                            objExcel.Application.Range(bRk2).Select
                            With objExcel.Application.Selection
                                .HorizontalAlignment = hAlignment
                                .VerticalAlignment = vAlignment
                                .WrapText = bWrapText
                            End With
                            With objExcel.Application.Selection.Font
                                .Name = sFont
                                .FontStyle = sFontStyle
                                .Size = dFontSize
                                .Underline = lUnderline
                                .ColorIndex = lColorIndex
                            End With
                            objExcel.Application.Selection.NumberFormat = sFormat
                            
                       objExcel.ActiveSheet.Cells((CLng(rssortgroup![Head_b_1]) + posAdd_H) + 1, ColLet2Num(rsGRReport![Letter])) = _
                       objExcel.ActiveSheet.Cells(CLng(rssortgroup![begin1]), ColLet2Num(rsGRReport![Add_Name_From_Data_H1_A]))
                       bRk2 = rsGRReport![Letter] & (CLng(rssortgroup![Head_b_1]) + posAdd_H) + 1 & ":" & _
                                rsGRReport![Letter] & (CLng(rssortgroup![Head_b_1]) + posAdd_H) + 1
                            objExcel.Application.Range(bRk2).Select
                            With objExcel.Application.Selection
                                .HorizontalAlignment = hAlignment
                                .VerticalAlignment = vAlignment
                                .WrapText = bWrapText_2
                            End With
                            With objExcel.Application.Selection.Font
                                .Name = sFont_2
                                .FontStyle = sFontStyle_2
                                .Size = dFontSize_2
                                .Underline = lUnderline_2
                                .ColorIndex = lColorIndex_2
                            End With
                            objExcel.Application.Selection.NumberFormat = sFormat_2
                        Else
                        'just use from data
                       objExcel.ActiveSheet.Cells((CLng(rssortgroup![Head_b_1]) + posAdd_H), ColLet2Num(rsGRReport![Letter])) = _
                       objExcel.ActiveSheet.Cells(CLng(rssortgroup![begin1]), ColLet2Num(rsGRReport![Add_Name_From_Data_H1_A]))
                       bRk2 = rsGRReport![Letter] & (CLng(rssortgroup![Head_b_1]) + posAdd_H) & ":" & _
                                rsGRReport![Letter] & (CLng(rssortgroup![Head_b_1]) + posAdd_H)
                            objExcel.Application.Range(bRk2).Select
                            With objExcel.Application.Selection
                                .HorizontalAlignment = hAlignment
                                .VerticalAlignment = vAlignment
                                .WrapText = bWrapText_2
                            End With
                            With objExcel.Application.Selection.Font
                                .Name = sFont_2
                                .FontStyle = sFontStyle_2
                                .Size = dFontSize_2
                                .Underline = lUnderline_2
                                .ColorIndex = lColorIndex_2
                            End With
                             objExcel.Application.Selection.NumberFormat = sFormat_2
                    End If
                Else
                 'use dHeadInsert
                       objExcel.ActiveSheet.Cells((rssortgroup![Head_b_1] + posAdd_H), ColLet2Num(rsGRReport![Letter])).Value = _
                       dHeadInsert
                       bRk2 = rsGRReport![Letter] & (CLng(rssortgroup![Head_b_1]) + posAdd_H) & ":" & _
                                rsGRReport![Letter] & (CLng(rssortgroup![Head_b_1]) + posAdd_H)
                            objExcel.Application.Range(bRk2).Select
                            With objExcel.Application.Selection
                                .HorizontalAlignment = hAlignment
                                .VerticalAlignment = vAlignment
                                .WrapText = bWrapText
                            End With
                            With objExcel.Application.Selection.Font
                                .Name = sFont
                                .FontStyle = sFontStyle
                                .Size = dFontSize
                                .Underline = lUnderline
                                .ColorIndex = lColorIndex
                            End With
                             objExcel.Application.Selection.NumberFormat = sFormat
               End If
                rsGRReport.MoveNext
            Wend
'----------------------------------------------------------------------------------------------------------------------------------------------------------
''---------------------END DATA RETRIVAL FOR SECTIONS FOR HEADER ------------------------------------------------------------------------------------------
'---------------------If they have mergeCells selected then do so - MERGE DATA FIRST -  SECTIONS-----------------------------------------------------------
''---------------------------------------------------------------------------------------------------------------------------------------------------------
      dHeadInsert = ""
      dHeadInsert2 = ""
                rsGRReport.MoveFirst
                While rsGRReport.EOF = False
                      If rsGRReport![Column_hide] = False Then
                        hBeginLetter = rsGRReport![Letter]
                        rsGRReport.MoveLast
                       End If
                    rsGRReport.MoveNext
                Wend
             rsgRR_Report_Desc.MoveFirst
             While rsgRR_Report_Desc.EOF = False
                 If rsgRR_Report_Desc![Column_hide] = False Then
                     hEndLetter = rsgRR_Report_Desc![Letter]
                     rsgRR_Report_Desc.MoveLast
                 End If
                 rsgRR_Report_Desc.MoveNext
             Wend
           rsGRReport.MoveFirst
           If rsGRReport![Merge_Center_H1] = True Then
            While rsGRReport.EOF = False
                'labels
                If rsGRReport![HeaderCol_H1_a] = True Then
                    posAdd_H = IIf(IsNull(rsGRReport![position_h1_a]), 0, rsGRReport![position_h1_a])
                    dHeadInsert = dHeadInsert & "  " & _
                    objExcel.ActiveSheet.Cells((CLng(rssortgroup![Head_b_1]) + posAdd_H), ColLet2Num(rsGRReport![Letter]))
                    objExcel.ActiveSheet.Cells((CLng(rssortgroup![Head_b_1]) + posAdd_H), ColLet2Num(rsGRReport![Letter])).ClearContents
                    'data
                    dHeadInsert2 = dHeadInsert2 & "  " & _
                    objExcel.ActiveSheet.Cells((CLng(rssortgroup![Head_b_1]) + posAdd_H + 1), ColLet2Num(rsGRReport![Letter]))
                    objExcel.ActiveSheet.Cells((CLng(rssortgroup![Head_b_1]) + posAdd_H + 1), ColLet2Num(rsGRReport![Letter])).ClearContents
                 End If
              rsGRReport.MoveNext
            Wend
            
            objExcel.ActiveSheet.Cells((CLng(rssortgroup![Head_b_1]) + posAdd_H), ColLet2Num(hBeginLetter)) = Trim(dHeadInsert)

Private Sub ApplyHeaderFontColor( _
    ByVal objTargetRange As Excel.Range, _
    ByVal varMetadataColor As Variant)

    Dim strColor As String

    strColor = UCase$(Trim$(Nz( _
        varMetadataColor, _
        vbNullString)))

    With objTargetRange.Font

        Select Case strColor
            Case vbNullString, "BLACK"
                .ColorIndex = 1
            Case "RED"
                .ColorIndex = 3
            Case "GREEN"
                .ColorIndex = 50
            Case "BLUE"
                .ColorIndex = 5
            Case "GOLD"
                .ColorIndex = 44
            Case "LIGHT GOLD"
                .ColorIndex = 36
            Case "XLTHEMECOLORDARK1"
                .ThemeColor = xlThemeColorDark1
            Case "XLTHEMECOLORLIGHT1"
                .ThemeColor = xlThemeColorLight1
            Case "XLTHEMECOLORDARK2"
                .ThemeColor = xlThemeColorDark2
            Case "XLTHEMECOLORLIGHT2"
                .ThemeColor = xlThemeColorLight2
            Case "XLTHEMECOLORACCENT1"
                .ThemeColor = xlThemeColorAccent1
            Case "XLTHEMECOLORACCENT2"
                .ThemeColor = xlThemeColorAccent2
            Case "XLTHEMECOLORACCENT3"
                .ThemeColor = xlThemeColorAccent3
            Case "XLTHEMECOLORACCENT4"
                .ThemeColor = xlThemeColorAccent4
            Case "XLTHEMECOLORACCENT5"
                .ThemeColor = xlThemeColorAccent5
            Case "XLTHEMECOLORACCENT6"
                .ThemeColor = xlThemeColorAccent6
            Case Else

                If IsNumeric(strColor) Then
                    .Color = CLng(strColor)
                Else
                    Debug.Print _
                        "Unrecognized header font color: "; _
                        strColor
                    .ColorIndex = 1
                End If
        End Select
    End With

End Sub
Private Sub ApplyHeaderFillColor( _
    ByVal objTargetRange As Excel.Range, _
    ByVal varMetadataColor As Variant)

    Dim strColor As String

    strColor = UCase$(Trim$(Nz( _
        varMetadataColor, _
        vbNullString)))

    With objTargetRange.Interior

        Select Case strColor

            '----------------------------------------------
            'No fill
            '----------------------------------------------
            Case vbNullString, _
                 "NONE", _
                 "XLCOLORINDEXNONE"

                .Pattern = xlSolid
                .ColorIndex = xlColorIndexNone

            '----------------------------------------------
            'Named dropdown values
            '----------------------------------------------
            Case "BLACK"

                .Pattern = xlSolid
                .ColorIndex = 1

            Case "RED"

                .Pattern = xlSolid
                .ColorIndex = 3

            Case "GREEN"

                .Pattern = xlSolid
                .ColorIndex = 50

            Case "BLUE"

                .Pattern = xlSolid
                .ColorIndex = 5

            Case "GOLD"

                .Pattern = xlSolid
                .ColorIndex = 44

            Case "LIGHT GOLD"

                .Pattern = xlSolid
                .ColorIndex = 36

            '----------------------------------------------
            'Theme constants stored as text
            '----------------------------------------------
            Case "XLTHEMECOLORDARK1"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorDark1
                .TintAndShade = 0

            Case "XLTHEMECOLORLIGHT1"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorLight1
                .TintAndShade = 0

            Case "XLTHEMECOLORDARK2"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorDark2
                .TintAndShade = 0

            Case "XLTHEMECOLORLIGHT2"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorLight2
                .TintAndShade = 0

            Case "XLTHEMECOLORACCENT1"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorAccent1
                .TintAndShade = 0

            Case "XLTHEMECOLORACCENT2"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorAccent2
                .TintAndShade = 0

            Case "XLTHEMECOLORACCENT3"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorAccent3
                .TintAndShade = 0

            Case "XLTHEMECOLORACCENT4"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorAccent4
                .TintAndShade = 0

            Case "XLTHEMECOLORACCENT5"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorAccent5
                .TintAndShade = 0

            Case "XLTHEMECOLORACCENT6"

                .Pattern = xlSolid
                .ThemeColor = xlThemeColorAccent6
                .TintAndShade = 0

            '----------------------------------------------
            'Numeric Excel color value
            '
            'Example: 65535 is yellow.
            '----------------------------------------------
            Case Else

                If IsNumeric(strColor) Then

                    .Pattern = xlSolid
                    .Color = CLng(strColor)

                Else

                    Debug.Print _
                        "Unrecognized header fill color: "; _
                        strColor

                    .ColorIndex = xlColorIndexNone

                End If

        End Select

        .PatternColorIndex = xlAutomatic

    End With

Private Function GetHeaderFontStyle( _
    ByVal varFontStyle As Variant) As String

    Select Case UCase$(Trim$(Nz( _
        varFontStyle, _
        vbNullString)))

        Case "REGULAR"

            GetHeaderFontStyle = "Regular"

        Case "BOLD"

            GetHeaderFontStyle = "Bold"

        Case "ITALIC"

            GetHeaderFontStyle = "Italic"

        Case "BOLD ITALIC"

            GetHeaderFontStyle = "Bold Italic"

        Case Else

            GetHeaderFontStyle = "Regular"

    End Select

End Function
Private Function GetHeaderLineWeight( _
    ByVal varLineWeight As Variant) As Long

    Select Case UCase$(Trim$(Nz( _
        varLineWeight, _
        vbNullString)))

        Case "THIN", "XLTHIN", "2"

            GetHeaderLineWeight = xlThin

        Case "MEDIUM", "XLMEDIUM", "-4138"

            GetHeaderLineWeight = xlMedium

        Case "THICK", "XLTHICK", "4"

            GetHeaderLineWeight = xlThick

        Case Else

            GetHeaderLineWeight = xlThin

    End Select

End Function
Private Sub ApplyHeaderBorders( _
    ByVal objTargetRange As Excel.Range, _
    ByVal lngLineWeight As Long, _
    ByVal blnTop As Boolean, _
    ByVal blnBottom As Boolean, _
    ByVal blnLeft As Boolean, _
    ByVal blnRight As Boolean, _
    ByVal blnInsideVertical As Boolean)

    If blnTop Then

        With objTargetRange.Borders(xlEdgeTop)

            .LineStyle = xlContinuous
            .Weight = lngLineWeight
            .ColorIndex = xlAutomatic

        End With

    End If

    If blnBottom Then

        With objTargetRange.Borders(xlEdgeBottom)

            .LineStyle = xlContinuous
            .Weight = lngLineWeight
            .ColorIndex = xlAutomatic

        End With

    End If

    If blnLeft Then

        With objTargetRange.Borders(xlEdgeLeft)

            .LineStyle = xlContinuous
            .Weight = lngLineWeight
            .ColorIndex = xlAutomatic

        End With

    End If

    If blnRight Then

        With objTargetRange.Borders(xlEdgeRight)

            .LineStyle = xlContinuous
            .Weight = lngLineWeight
            .ColorIndex = xlAutomatic

        End With

    End If

    If blnInsideVertical Then

        If objTargetRange.Columns.Count > 1 Then

            If objTargetRange.MergeCells = False Then

                With objTargetRange.Borders( _
                    xlInsideVertical)

                    .LineStyle = xlContinuous
                    .Weight = lngLineWeight
                    .ColorIndex = xlAutomatic

                End With

            End If

        End If

    End If

End Sub

                                rsEditNums![head_b_1_temp] = rsEditNums![head_b_1_temp] - prpHead_2
                            rsEditNums.Update
                            rsTemp.Edit
                                rsTemp![Head_b_1] = rsEditNums![head_b_1_temp]

                      rsEditNums.MoveNext

                rsTemp.MoveNext
           Wend
        rsStrNameHolder.MoveNext
    Wend

    If i = 1 Then
     fn_FindBreaks = True
         Exit Function
    End If
   
  '--------------------------------------------------------------------------------------------------

             'need to move to the last query/table in rsstrtable - prpSortingGrouping
  End If
             
           
 'try to free up some memory
Set rs = Nothing
Set rsTemp = Nothing

    fn_FindBreaks = True
'leave
Exit Function

End Function

Function ColLet2Num(ColumnLetter As String) As Long
Dim CL As String, CN As Long, n As Integer, S As Long
Dim Se As Integer, c As Integer, T As String, ti As Integer, A As Long

Let CL = ColumnLetter
Let n = Len(CL)
Let S = 0
    For Se = 0 To n - 1
        Let S = S + 26 ^ Se
    Next Se
    Let T = ""
    Let A = 0
    For c = 1 To n
            T = UCase(Mid(CL, c, 1))
            Let ti = 65
            Do Until Chr(ti) = T
                Let ti = ti + 1
                If ti > 90 Then
                    Let CN = "Error"
                    GoTo 1
                End If
            Loop
        Let A = A + ((ti - 65) * (26 ^ (n - c)))
    Next
    Let CN = S + A
1     ColLet2Num = CN
End Function

Private Sub CreateMedicareTop10Chart(ByRef rsStrTable As DAO.Recordset, ByVal strTabName As String)

    Const TOP_COUNT As Long = 10

    'Current Medicare demo grand-total columns.
    Const CATEGORY_COLUMN As String = "I"
    Const PAYMENT_COLUMN As String = "N"

    Const CHART_SHEET_NAME As String = "Medicare_Chart"
    Const CHART_OBJECT_NAME As String = "chtMedicareTop10"

    Dim db As DAO.Database
    Dim rsGroupSort As DAO.Recordset

    Dim objSourceSheet As Excel.Worksheet
    Dim objChartSheet As Excel.Worksheet
    Dim objChartObject As Excel.ChartObject
    Dim objChart As Excel.Chart
    Dim objSeries As Excel.Series

    Dim strSQL As String
    Dim strCategory As String

    Dim lngBeginRow As Long
    Dim lngEndRow As Long
    Dim lngCurrentRow As Long
    Dim lngPosition As Long
    Dim lngShift As Long
    Dim lngOutputRow As Long
    Dim lngActualCount As Long

    Dim dblPayment As Double

    Dim arrCategory(1 To TOP_COUNT) As String
    Dim arrPayment(1 To TOP_COUNT) As Double

    On Error GoTo ErrorHandler

    '---------------------------------------------------------
    'This chart applies only to the Medicare demonstration.
    'It is safe for OpenXL to call this for every worksheet.
    '---------------------------------------------------------
    If strTabName <> "Medicare_Report" Then
        Exit Sub
    End If

    Set db = DBEngine(0)(0)

    '---------------------------------------------------------
    'Find the Grand Totals band that fn_FindBreaks previously
    'calculated and stored in the sorting/grouping table.
    '---------------------------------------------------------
    strSQL = _
        "SELECT * " & _
        "FROM [" & prpSortingGrouping & "] " & _
        "WHERE [" & prpSortingGrouping & "].[Level] = """ & _
            Replace(rsStrTable!queryname, """", """""") & """ " & _
        "AND [Description1] LIKE '*Grand Totals*';"

    Debug.Print strSQL

    Set rsGroupSort = db.OpenRecordset( _
        strSQL, _
        dbOpenSnapshot)

    If rsGroupSort.BOF And rsGroupSort.EOF Then

        Err.Raise _
            vbObjectError + 3100, _
            "CreateMedicareTop10Chart", _
            "The Medicare Grand Totals band was not found."

    End If

    rsGroupSort.MoveFirst

    lngBeginRow = CLng(rsGroupSort!begin1)
    lngEndRow = CLng(rsGroupSort!End1)

    Debug.Print _
        "Medicare Grand Totals band: rows "; _
        lngBeginRow; _
        " through "; _
        lngEndRow

    Set objSourceSheet = _
        objExcel.ActiveWorkbook.Worksheets(strTabName)

    '---------------------------------------------------------
    'Calculate the finished Grand Totals band before reading
    'its formula results. OpenXL may currently be operating
    'with Excel calculation set to manual.
    '---------------------------------------------------------
    objSourceSheet.Range( _
        CATEGORY_COLUMN & lngBeginRow & ":" & _
        PAYMENT_COLUMN & lngEndRow).Calculate

    '=========================================================
    'READ THE GRAND-TOTAL BAND AND RETAIN THE TOP TEN
    '
    'The report itself remains in its original sequence.
    'No worksheet sorting and no temporary Access table are
    'required.
    '=========================================================
    For lngCurrentRow = lngBeginRow To lngEndRow

        strCategory = Trim$(CStr(Nz( _
            objSourceSheet.Range( _
                CATEGORY_COLUMN & lngCurrentRow).Value, _
            vbNullString)))

        If Len(strCategory) > 0 Then

            If IsNumeric(objSourceSheet.Range( _
                PAYMENT_COLUMN & lngCurrentRow).Value) Then

                dblPayment = CDbl( _
                    objSourceSheet.Range( _
                        PAYMENT_COLUMN & lngCurrentRow).Value)

                '---------------------------------------------
                'Insert the value into its descending Top 10
                'position.
                '---------------------------------------------
                For lngPosition = 1 To TOP_COUNT

                    If dblPayment > arrPayment(lngPosition) Then

                        'Shift lower-ranking entries down.
                        For lngShift = TOP_COUNT To _
                            lngPosition + 1 Step -1

                            arrCategory(lngShift) = _
                                arrCategory(lngShift - 1)

                            arrPayment(lngShift) = _
                                arrPayment(lngShift - 1)

                        Next lngShift

                        arrCategory(lngPosition) = _
                            strCategory

                        arrPayment(lngPosition) = _
                            dblPayment

                        Exit For

                    End If
                Next lngPosition
            End If
        End If

    Next lngCurrentRow

    '---------------------------------------------------------
    'Count the populated results.
    '---------------------------------------------------------
    For lngPosition = 1 To TOP_COUNT

        If Len(arrCategory(lngPosition)) > 0 Then
            lngActualCount = lngActualCount + 1
        End If

    Next lngPosition

    If lngActualCount = 0 Then

        Err.Raise _
            vbObjectError + 3101, _
            "CreateMedicareTop10Chart", _
            "No numeric Medicare payment values were found."

    End If

    '=========================================================
    'RECREATE THE MEDICARE CHART WORKSHEET
    '=========================================================
    objExcel.Application.DisplayAlerts = False

    On Error Resume Next

    objExcel.ActiveWorkbook.Worksheets( _
        CHART_SHEET_NAME).Delete

    On Error GoTo ErrorHandler

    objExcel.Application.DisplayAlerts = True

    Set objChartSheet = _
        objExcel.ActiveWorkbook.Worksheets.Add( _
            After:=objExcel.ActiveWorkbook.Worksheets( _
                objExcel.ActiveWorkbook.Worksheets.Count))

    objChartSheet.Name = CHART_SHEET_NAME

    '=========================================================
    'WRITE THE COMPACT CHART-SOURCE TABLE
    '=========================================================
    With objChartSheet

        .Range("A1").Value = _
            "2024 Texas Medicare"

        .Range("A2").Value = _
            "Top 10 Categories by Estimated Medicare Payment"

        .Range("A4").Value = _
            "RBCS Category"

        .Range("B4").Value = _
            "Estimated Medicare Payment ($ Millions)"

        '-----------------------------------------------------
        'Main chart-sheet title.
        '-----------------------------------------------------
        With .Range("A1:B1")

            .Merge

            .Interior.Color = _
                rgb(31, 78, 121)

            With .Font
                .Color = rgb(255, 255, 255)
                .Bold = True
                .Size = 18
            End With

            .HorizontalAlignment = xlLeft
            .VerticalAlignment = xlCenter

        End With

        '-----------------------------------------------------
        'Subtitle.
        '-----------------------------------------------------
        With .Range("A2:B2")

            .Merge

            With .Font
                .Italic = True
                .Size = 11
            End With

        End With

        '-----------------------------------------------------
        'Source-table headings.
        '-----------------------------------------------------
        With .Range("A4:B4")

            .Interior.Color = _
                rgb(47, 117, 181)

            With .Font
                .Color = rgb(255, 255, 255)
                .Bold = True
            End With

            .HorizontalAlignment = xlCenter
            .VerticalAlignment = xlCenter

        End With

        '-----------------------------------------------------
        'Write the ten highest payment categories.
        '
        'Values are stored in millions so the chart labels and
        'axis remain concise.
        '-----------------------------------------------------
        For lngPosition = 1 To lngActualCount

            lngOutputRow = lngPosition + 4

            .Cells(lngOutputRow, 1).Value = _
                arrCategory(lngPosition)

            .Cells(lngOutputRow, 2).Value = _
                arrPayment(lngPosition) / 1000000#

            .Cells(lngOutputRow, 2).NumberFormat = _
                "$#,##0.0"

            If lngPosition Mod 2 = 0 Then

                .Range( _
                    "A" & lngOutputRow & _
                    ":B" & lngOutputRow).Interior.Color = _
                        rgb(217, 234, 247)

            End If

        Next lngPosition

        .Columns("A").ColumnWidth = 56
        .Columns("B").ColumnWidth = 37.86

        .Rows("1:1").RowHeight = 30
        .Rows("4:4").RowHeight = 30

    End With

    '=========================================================
    'CREATE THE EMBEDDED HORIZONTAL BAR CHART
    '=========================================================
    Set objChartObject = _
        objChartSheet.ChartObjects.Add( _
            Left:=objChartSheet.Range("D2").Left, _
            Top:=objChartSheet.Range("D2").Top, _
            Width:=820, _
            Height:=480)

    objChartObject.Name = CHART_OBJECT_NAME

    Set objChart = objChartObject.Chart

    With objChart

        'Horizontal clustered bar chart.
        .ChartType = xlBarClustered

        .HasTitle = True

        .ChartTitle.Text = _
            "Top 10 Categories by Estimated Medicare Payment"

        .HasLegend = False

        'Remove any automatically created series.
        Do While .SeriesCollection.Count > 0
            .SeriesCollection(1).Delete
        Loop

        Set objSeries = _
            .SeriesCollection.NewSeries

        With objSeries

            .Name = _
                "=""Estimated Medicare Payment"""

            .XValues = objChartSheet.Range( _
                "A5:A" & 4 + lngActualCount)

            .Values = objChartSheet.Range( _
                "B5:B" & 4 + lngActualCount)

            .Format.Fill.ForeColor.rgb = _
                rgb(91, 155, 213)

            .Format.Line.ForeColor.rgb = _
                rgb(31, 78, 121)

            .ApplyDataLabels

            .DataLabels.NumberFormat = _
                "$0.0"

            .DataLabels.Position = _
                xlLabelPositionOutsideEnd

        End With

        'The array is descending. Reverse the category axis so
        'the largest category displays at the top.
        .Axes(xlCategory).ReversePlotOrder = True

        With .Axes(xlCategory)

            .HasTitle = True

            .AxisTitle.Text = _
                "RBCS Category"

            .TickLabels.Font.Size = 9

        End With

        With .Axes(xlValue)

            .HasTitle = True

            .AxisTitle.Text = _
                "Estimated Medicare Payment ($ Millions)"

            .TickLabels.NumberFormat = "$0"

            .HasMajorGridlines = True

        End With

        .ChartArea.Format.Fill.ForeColor.rgb = _
            rgb(255, 255, 255)

        .PlotArea.Format.Fill.ForeColor.rgb = _
            rgb(255, 255, 255)

    End With

    '---------------------------------------------------------
    'Leave the new chart sheet in a clean opening position.
    '---------------------------------------------------------
    objChartSheet.Activate
    objChartSheet.Range("A1").Select

ExitProcedure:

    On Error Resume Next

    objExcel.Application.DisplayAlerts = True

    If Not rsGroupSort Is Nothing Then
        rsGroupSort.Close
    End If

    Set objSeries = Nothing
    Set objChart = Nothing
    Set objChartObject = Nothing
    Set objChartSheet = Nothing
    Set objSourceSheet = Nothing
    Set rsGroupSort = Nothing
    Set db = Nothing

    Exit Sub

ErrorHandler:

    Dim lngErrorNumber As Long
    Dim strErrorDescription As String

    lngErrorNumber = Err.Number
    strErrorDescription = Err.Description

    objExcel.Application.DisplayAlerts = True

    Err.Raise _
        lngErrorNumber, _
        "CreateMedicareTop10Chart", _
        strErrorDescription

End Sub
